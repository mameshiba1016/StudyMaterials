cmake_minimum_required(VERSION 3.20)
project(cpp_scope_storage_lifetime LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/W4 /permissive- /utf-8)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_library(lifetime_trace STATIC src/LifetimeTrace.cpp)
target_include_directories(lifetime_trace PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
add_executable(scope_storage_lifetime src/main.cpp)
target_link_libraries(scope_storage_lifetime PRIVATE lifetime_trace)
include(CTest)
if(BUILD_TESTING)
  add_executable(scope_storage_lifetime_tests tests/LifetimeTraceTests.cpp)
  target_link_libraries(scope_storage_lifetime_tests PRIVATE lifetime_trace)
  add_test(NAME scope_storage_lifetime_tests COMMAND scope_storage_lifetime_tests)
endif()
