cmake_minimum_required(VERSION 3.20)
project(cpp_functions LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/W4 /permissive- /utf-8)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_library(damage_functions STATIC src/DamageFunctions.cpp)
target_include_directories(damage_functions PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
add_executable(functions src/main.cpp)
target_link_libraries(functions PRIVATE damage_functions)
include(CTest)
if(BUILD_TESTING)
  add_executable(functions_tests tests/DamageFunctionsTests.cpp)
  target_link_libraries(functions_tests PRIVATE damage_functions)
  add_test(NAME functions_tests COMMAND functions_tests)
endif()
