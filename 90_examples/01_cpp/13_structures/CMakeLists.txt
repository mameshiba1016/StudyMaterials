cmake_minimum_required(VERSION 3.20)
project(cpp_structures LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/W4 /permissive- /utf-8)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_library(combat_data STATIC src/CombatData.cpp)
target_include_directories(combat_data PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
add_executable(structures src/main.cpp)
target_link_libraries(structures PRIVATE combat_data)
include(CTest)
add_executable(structures_tests tests/CombatDataTests.cpp)
target_link_libraries(structures_tests PRIVATE combat_data)
add_test(NAME structures_tests COMMAND structures_tests)
