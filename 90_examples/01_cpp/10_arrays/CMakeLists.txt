cmake_minimum_required(VERSION 3.20)
project(cpp_arrays LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
 add_compile_options(/W4 /permissive- /utf-8)
endif()
add_library(array_rules STATIC src/ArrayRules.cpp)
target_include_directories(array_rules PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
add_executable(arrays src/main.cpp)
target_link_libraries(arrays PRIVATE array_rules)
include(CTest)
add_executable(arrays_tests tests/ArrayRulesTests.cpp)
target_link_libraries(arrays_tests PRIVATE array_rules)
add_test(NAME arrays_tests COMMAND arrays_tests)
