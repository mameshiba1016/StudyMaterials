cmake_minimum_required(VERSION 3.20)
project(cpp_strings_encoding LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/W4 /permissive- /utf-8)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_library(text_rules STATIC src/TextRules.cpp)
target_include_directories(text_rules PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
add_executable(strings_encoding src/main.cpp)
target_link_libraries(strings_encoding PRIVATE text_rules)
include(CTest)
add_executable(strings_encoding_tests tests/TextRulesTests.cpp)
target_link_libraries(strings_encoding_tests PRIVATE text_rules)
add_test(NAME strings_encoding_tests COMMAND strings_encoding_tests)
