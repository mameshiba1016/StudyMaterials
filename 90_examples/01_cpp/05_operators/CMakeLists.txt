cmake_minimum_required(VERSION 3.20)
project(cpp_operators VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
    add_compile_options(/W4 /permissive- /utf-8)
else()
    add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_library(action_rules STATIC src/ActionRules.cpp)
target_include_directories(action_rules PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
add_executable(operators src/main.cpp)
target_link_libraries(operators PRIVATE action_rules)
include(CTest)
if(BUILD_TESTING)
    add_executable(operators_tests tests/ActionRulesTests.cpp)
    target_link_libraries(operators_tests PRIVATE action_rules)
    add_test(NAME operators_tests COMMAND operators_tests)
endif()
