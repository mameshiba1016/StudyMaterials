cmake_minimum_required(VERSION 3.20)
project(cpp_comments_and_intent VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
    add_compile_options(/W4 /permissive- /utf-8)
else()
    add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_library(cooldown STATIC src/Cooldown.cpp)
target_include_directories(cooldown PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
add_executable(comments_and_intent src/main.cpp)
target_link_libraries(comments_and_intent PRIVATE cooldown)
include(CTest)
if(BUILD_TESTING)
    add_executable(comments_and_intent_tests tests/CooldownTests.cpp)
    target_link_libraries(comments_and_intent_tests PRIVATE cooldown)
    add_test(NAME comments_and_intent_tests COMMAND comments_and_intent_tests)
endif()
