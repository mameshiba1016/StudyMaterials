cmake_minimum_required(VERSION 3.20)
project(cpp_conditionals LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/W4 /permissive- /utf-8)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_library(action_decision STATIC src/ActionDecision.cpp)
target_include_directories(action_decision PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
add_executable(conditionals src/main.cpp)
target_link_libraries(conditionals PRIVATE action_decision)
include(CTest)
if(BUILD_TESTING)
  add_executable(conditionals_tests tests/ActionDecisionTests.cpp)
  target_link_libraries(conditionals_tests PRIVATE action_decision)
  add_test(NAME conditionals_tests COMMAND conditionals_tests)
endif()
