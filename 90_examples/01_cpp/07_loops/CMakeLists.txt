cmake_minimum_required(VERSION 3.20)
project(cpp_loops LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/W4 /permissive- /utf-8)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_library(enemy_loop STATIC src/EnemyLoop.cpp)
target_include_directories(enemy_loop PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
add_executable(loops src/main.cpp)
target_link_libraries(loops PRIVATE enemy_loop)
include(CTest)
if(BUILD_TESTING)
  add_executable(loops_tests tests/EnemyLoopTests.cpp)
  target_link_libraries(loops_tests PRIVATE enemy_loop)
  add_test(NAME loops_tests COMMAND loops_tests)
endif()
